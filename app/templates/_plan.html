{#- Turns the plain-text plan into day cards. Put this file next to base.html. -#} {% macro render_plan(text) -%} {%- set ns = namespace(day=false, list=false) -%}
{%- for raw in (text or '').splitlines() -%} {%- set s = raw.strip() -%} {%- set line = s.replace('**', '').replace('__', '').lstrip('# ').strip() -%} {%- if line -%} {%- if line.lower().startswith('day ') and line[4:5].isdigit() -%} {%- if ns.list %}{% set ns.list = false %}{% endif -%} {%- if ns.day %}{% endif -%} {%- set ns.day = true -%} {%- set sep = ' - ' if ' - ' in line else (' – ' if ' – ' in line else ': ') -%} {%- set parts = line.split(sep, 1) %}
{{ parts[0].rstrip(':') }} {%- if parts|length > 1 %}

{{ parts[1] }}

{% endif %}
{%- elif s.startswith(('- ', '• ', '* ')) -%} {%- if not ns.list %}
    {% set ns.list = true %}{% endif -%} {%- set item = line[2:].strip() %}
  • {%- if ':' in item %}{% set bits = item.split(':', 1) %}{{ bits[0] }}{{ bits[1].strip() }} {%- else %}{{ item }}{% endif -%}
    • {%- else -%} {%- if ns.list %}
{% set ns.list = false %}{% endif -%} {%- if line.endswith(':') %}

{{ line.rstrip(':') }}

{%- elif ':' in line[:48] %}{% set bits = line.split(':', 1) %}

{{ bits[0] }}: {{ bits[1].strip() }}

{%- else %}

{{ line }}

{%- endif -%} {%- endif -%} {%- endif -%} {%- endfor -%} {%- if ns.list %}{% endif -%} {%- if ns.day %}
{% endif %}
{%- endmacro %}
